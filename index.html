// index.html ke script section mein onScanSuccess ko is se badal dein function onScanSuccess(decodedText) { document.getElementById('result').style.display = "block"; // 1. Pehle Mobile Memory (Local Storage) mein check karo const localData = localStorage.getItem('mart_products'); if (localData) { const products = JSON.parse(localData); // Find product by ID const item = products.find(p => p.id.toString().trim() === decodedText.trim()); if (item) { console.log("Offline Data Found!"); showProduct(item); return; // Internet pe janay ki zaroorat nahi } } // 2. Agar mobile mein nahi mila, to Internet (Live Sheet) se check karo fetch(API + "?action=getDetails&id=" + decodedText) .then(res => res.json()) .then(data => { if (!data.error) { showProduct(data); } else { showNotFound(decodedText); } }) .catch(err => { document.getElementById('p-en').innerText = "Offline & Not in Cache"; }); } function showProduct(data) { document.getElementById('p-en').innerText = data.en || data.enName; document.getElementById('p-ur').innerText = data.ur || data.urName || ""; document.getElementById('p-price').innerText = "Rs. " + (data.price || data.pr); document.getElementById('p-unit').innerText = data.unit || data.un || "Per Item"; document.getElementById('p-img').src = (data.img && data.img.length > 5) ? data.img : "https://via.placeholder.com/400?text=No+Image"; }
